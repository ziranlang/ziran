cmake_minimum_required(VERSION 3.16)
project(zxcompare C ASM)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WAMR_ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../build/deps/wasm-micro-runtime-WAMR-2.4.5")
set(WAMR_BUILD_INTERP 0)
set(WAMR_BUILD_AOT 1)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 0)
if(ANDROID)
  set(WAMR_BUILD_PLATFORM android)
  set(WAMR_BUILD_TARGET ARMV7)
else()
  set(WAMR_BUILD_PLATFORM linux)
  set(WAMR_BUILD_TARGET X86_64)
endif()
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(wamr STATIC ${WAMR_RUNTIME_LIB_SOURCE})
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(SOURCES engine.c component.c ../zx_compare_control.c)
if(ANDROID)
  add_library(zxcompare SHARED ${SOURCES} activity.c visual.c)
  target_link_libraries(zxcompare PRIVATE wamr ${ROOT}/build/android-host/armeabi-v7a/libziran.a android log jnigraphics m dl)
else()
  add_executable(zxcompare ${SOURCES} cli.c)
  target_link_libraries(zxcompare PRIVATE wamr ${ROOT}/build/libziran.a m dl pthread)
endif()
target_compile_options(zxcompare PRIVATE -O3 -Wall -Wextra)
target_include_directories(zxcompare PRIVATE ${ROOT}/include ${WAMR_ROOT_DIR}/core/iwasm/include)
