Learn / Examples

See Ziran in action.

These examples are complete files in the repository, and the output shown is what they print. Build the toolchain with make, then run the commands from the repository root. For an entry that returns a value, the portable runner prints the result.

02 / greet.zi

Print values

Each % in a print format takes the next argument, and %% prints a percent sign.

main :: () {
    name := "Ziran";
    year: s32 = 2026;
    print("Hello, %! It is %.\n", name, year);
    print("% + % = %\n", 40, 2, 40 + 2);
    print("pi is about %, and 100%% sure\n", 3.14159);
}
build/bin/ziran bundle --root . --entry greet:main -o greet.zib site/examples/greet.zi
build/bin/ziran run greet.zib
# Hello, Ziran! It is 2026.
# 40 + 2 = 42
# pi is about 3.14159, and 100% sure

05 / functions.zi

Functions

Plain calls, default and named arguments, and recursion.

Add :: (a: s32, b: s32) -> s32 {
    return a + b;
}

Greet :: (name: string, greeting: string = "Hello") {
    print("%, %!\n", greeting, name);
}

Factorial :: (n: s64) -> s64 {
    if n <= 1 { return 1; }
    return n * Factorial(n - 1);
}

Fibonacci :: (n: s32) -> s32 {
    if n < 2 { return n; }
    return Fibonacci(n - 1) + Fibonacci(n - 2);
}

main :: () {
    print("1 + 2 = %\n", Add(1, 2));
    Greet("Ziran");
    Greet("world", greeting = "Goodbye");
    print("10! = %\n", Factorial(10));
    print("fibonacci(20) = %\n", Fibonacci(20));
}
build/bin/ziran bundle --root . --entry functions:main -o functions.zib site/examples/functions.zi
build/bin/ziran run functions.zib
# 1 + 2 = 3
# Hello, Ziran!
# Goodbye, world!
# 10! = 3628800
# fibonacci(20) = 6765

06 / conditions.zi

Conditions

If and else chains, and an if-case that picks a branch by value.

Describe :: (number: s32) -> string {
    if number < 0 {
        return "negative";
    } else if number == 0 {
        return "zero";
    }
    return "positive";
}

DayKind :: (day: s32) -> string {
    if day == {
        case 6; return "weekend";
        case 7; return "weekend";
    }
    return "weekday";
}

main :: () {
    print("-4 is %\n", Describe(-4));
    print("0 is %\n", Describe(0));
    print("9 is %\n", Describe(9));

    if 7 % 2 == 0 {
        print("7 is even\n");
    } else {
        print("7 is odd\n");
    }

    print("day 3 is a %\n", DayKind(3));
    print("day 6 is a %\n", DayKind(6));
}
build/bin/ziran bundle --root . --entry conditions:main -o conditions.zib site/examples/conditions.zi
build/bin/ziran run conditions.zib
# -4 is negative
# 0 is zero
# 9 is positive
# 7 is odd
# day 3 is a weekday

07 / loops.zi

Loops

While, ranges, collections, break, and continue.

main :: () {
    index: s32 = 1;
    while index <= 3 {
        print("while %\n", index);
        index += 1;
    }

    for step: 0..2 {
        print("range %\n", step);
    }

    fruits := string.["apple", "banana", "cherry"];
    for fruit, position: fruits {
        print("% is at %\n", fruit, position);
    }

    for number: 0..9 {
        if number % 2 == 0 { continue; }
        if number > 7 { break; }
        print("odd %\n", number);
    }
}

08 / arrays.zi

Arrays and slices

A fixed array, a view into part of it, and a sum.

Sum :: (numbers: []s32) -> s32 {
    total: s32 = 0;
    for numbers { total += it; }
    return total;
}

main :: () {
    empty: [5]s32;
    print("empty: % % % % %\n", empty[0], empty[1], empty[2], empty[3], empty[4]);

    numbers := s32.[1, 2, 3, 4, 5];
    numbers[4] = 100;
    print("set: %, count: %\n", numbers[4], numbers.count);

    middle := numbers[1:4];
    print("middle has % items starting with %\n", middle.count, middle[0]);

    print("sum of all: %\n", Sum(numbers[:]));
    print("sum of middle: %\n", Sum(middle));
}

build/bin/ziran run arrays.zib

09 / structs.zi

Structs

Records with fields, passed and returned by value.

Point :: struct {
    x: s32;
    y: s32;
}

Rectangle :: struct {
    origin: Point;
    width: s32;
    height: s32;
}

Area :: (shape: Rectangle) -> s32 {
    return shape.width * shape.height;
}

Grow :: (shape: Rectangle, amount: s32) -> Rectangle {
    grown := shape;
    grown.width += amount;
    grown.height += amount;
    return grown;
}

main :: () {
    corner := Point.{x = 3, y = 4};
    print("point: (%, %)\n", corner.x, corner.y);

    box := Rectangle.{origin = corner, width = 10, height = 5};
    print("area: %\n", Area(box));

    box = Grow(box, 2);
    print("grown: %x% with area %\n", box.width, box.height, Area(box));

    copy := box;
    copy.width = 1;
    print("the copy is % wide, the original still %\n", copy.width, box.width);
}
build/bin/ziran bundle --root . --entry structs:main -o structs.zib site/examples/structs.zi
build/bin/ziran run structs.zib
# point: (3, 4)
# area: 50
# grown: 12x7 with area 84
# the copy is 1 wide, the original still 12

11 / text_basics.zi

Strings

Length, bytes, slices, and comparison.

CountByte :: (text: string, wanted: u8) -> s32 {
    found: s32 = 0;
    for byte: text {
        if byte == wanted { found += 1; }
    }
    return found;
}

main :: () {
    word := "banana";
    print("% has % bytes\n", word, word.count);
    print("the first byte is %\n", word[0]);
    print("it contains % a's\n", CountByte(word, #char "a"));
    print("its middle is %\n", word[1:5]);
    print("equal to \"banana\": %\n", word == "banana");
    print("equal to \"apple\": %\n", word == "apple");
}

12 / defer.zi

Defer

A deferred statement runs when its scope ends, last deferred first.

Work :: (name: string) {
    print("open %\n", name);
    defer print("close %\n", name);
    print("use %\n", name);
}

main :: () {
    defer print("done\n");
    Work("file");
    {
        defer print("leave the block\n");
        print("inside the block\n");
    }
}

14 / primes.zi

The Sieve of Eratosthenes

Cross out every multiple of each prime.

LIMIT :: 50;

main :: () {
    composite: [LIMIT + 1]bool;
    for number: 2..LIMIT {
        if composite[number] { continue; }
        print("% ", number);
        multiple := number * number;
        while multiple <= LIMIT {
            composite[multiple] = true;
            multiple += number;
        }
    }
    print("\n");
}

build/bin/ziran run primes.zib

17 / Native output

Generate native code

Every program above prints the same output on the portable runner and as C, C++, Go, Rust, and Python; tests/examples.sh checks it. For Rust, Go, and Python, --exe with an entry writes a runnable program. Go, Cargo, or Python 3.10 is needed only to run that output.

build/bin/ziran build --target=rust --exe --entry fizzbuzz:main --root . -o out/rust site/examples/fizzbuzz.zi
cargo run --manifest-path out/rust/Cargo.toml

build/bin/ziran build --target=go --pkg main --exe --entry fizzbuzz:main --root . -o out/go site/examples/fizzbuzz.zi
(cd out/go && go run .)

build/bin/ziran build --target=py --exe --entry fizzbuzz:main --root . -o out/py site/examples/fizzbuzz.zi
python3 out/py

C and C++ output mirrors the source path, here out/c/site/examples/fizzbuzz.c and its header, which declares fizzbuzz_main; call it from your program, or put #program_export before main to make the output a complete program. --target=plan9-c writes experimental Plan 9 C.

build/bin/ziran build --target=c --root . -o out/c site/examples/fizzbuzz.zi
build/bin/ziran build --target=cpp --root . -o out/cpp site/examples/fizzbuzz.zi

Native and portable support can differ by feature. See the current status before choosing a target.
